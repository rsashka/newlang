117uint8_t HexToByte(const char c) {
118 if (c >= '0' && c <= '9') {
119 return c - 0x30;
120 }
121 if (c >= 'a' && c <= 'f') {
122 return c - 0x61 + 10;
123 }
124 if (c >= 'A' && c <= 'F') {
125 return c - 0x41 + 10;
126 }
127 LOG_RUNTIME("the symbol '%c' is not a hex digit!", c);

130size_t HexToBin(const char * str, uint8_t * buf, const size_t size) {
